# 03. データモデル

## 1. 方針

- **メモリ上の構造体が「正」**。SQLiteは永続化と記憶検索のための保存先。
- 位置や欲求のような高頻度更新値はDBに毎tick書かない。30秒ごとのスナップショットでまとめて保存する。
- 記憶（events / memories）は追記型。**書いた瞬間にDBへ**入れる（ペットの記憶が消えるのが一番痛い）。

## 2. メモリ上の型（`packages/shared/src/types.ts`）

```ts
export type EntityId = number;          // 島内で一意な連番
export type PlayerId = string;          // uuid v4

export type Vec2 = { x: number; y: number };   // タイル単位のfloat座標

export type TimeOfDay = 'morning' | 'day' | 'evening' | 'night';
export type Season   = 'spring' | 'summer' | 'autumn' | 'winter';
export type Weather  = 'clear' | 'cloudy' | 'rain' | 'fog';

/** 動物住民・ペット・プレイヤーアバターの共通形 */
export interface Actor {
  id: EntityId;
  kind: 'critter' | 'pet' | 'player';
  species: string;                 // 'rabbit' | 'mofi' | ... （アセット名と一致）
  name: string;
  pos: Vec2;
  facing: 'n' | 'e' | 's' | 'w';
  speed: number;                   // タイル/秒
  anim: 'idle' | 'walk' | 'act' | 'sleep' | 'talk';

  needs: Needs;                    // playerは未使用
  traits: Traits;
  ageDays: number;                 // 島日
  lifespanDays: number;
  health: number;                  // 0..100

  action: ActiveAction | null;      // 実行中の行動
  path: Vec2[] | null;             // 経路探索の結果

  // ペットのみ
  ownerId?: PlayerId;
  affection?: number;              // 懐き度 0..100
  intent?: PetIntent | null;       // LLMが決めた目標
}

export interface Needs {           // 0(満たされている)..100(切迫)
  hunger: number;
  sleep: number;
  social: number;
  safety: number;
  curiosity: number;
}

/** 性格。Big-Five風。0..1。遺伝する */
export interface Traits {
  energy: number;      // 活発さ → 移動頻度・速度
  sociability: number; // 交流行動の重み
  caution: number;     // 逃走しやすさ・探索範囲
  gluttony: number;    // 空腹の増加速度と採食重み
  curiosity: number;   // 未訪問タイルへの誘引
}

export type ActionKind =
  | 'idle' | 'wander' | 'goto'
  | 'eat' | 'drink' | 'sleep'
  | 'socialize' | 'flee' | 'nest'
  | 'harvest' | 'water' | 'pet'      // プレイヤー/ペットのインタラクト
  | 'talk' | 'follow' | 'explore' | 'help';

export interface ActiveAction {
  kind: ActionKind;
  targetEntity?: EntityId;
  targetTile?: Vec2;
  startedAtTick: number;
  durationTicks: number;
}

/** LLMが返す「次にやりたいこと」 */
export interface PetIntent {
  goal: 'follow_owner' | 'explore' | 'visit_friend' | 'gather'
      | 'help_critter' | 'rest' | 'watch_stars' | 'talk_to';
  targetEntity?: EntityId;
  targetTile?: Vec2;
  reason: string;        // 1文。デバッグ表示と日記の材料
  sayNow?: string;       // 直後に吹き出しで出すひとこと（任意）
  expiresAtTick: number;
}
```

## 3. SQLite スキーマ（`packages/server/src/db/schema.sql`）

```sql
PRAGMA journal_mode = WAL;
PRAGMA synchronous = NORMAL;

-- 島インスタンスの状態（1行 = 1島）
CREATE TABLE IF NOT EXISTS island (
  id            TEXT PRIMARY KEY,          -- 'main' など
  seed          TEXT NOT NULL,
  tick          INTEGER NOT NULL,          -- 累積tick（4Hz）
  island_day    INTEGER NOT NULL,
  season        TEXT NOT NULL,
  weather       TEXT NOT NULL,
  updated_at    INTEGER NOT NULL           -- unix ms（catch-up判定に使う）
);

-- 30秒ごとのスナップショット（動物・資源・設置物をまとめてJSONで）
CREATE TABLE IF NOT EXISTS island_snapshot (
  island_id     TEXT PRIMARY KEY,
  tick          INTEGER NOT NULL,
  critters_json TEXT NOT NULL,
  resources_json TEXT NOT NULL,
  placeables_json TEXT NOT NULL,
  tiles_decay   BLOB NOT NULL               -- 128*128 の Uint8Array
);

CREATE TABLE IF NOT EXISTS player (
  id            TEXT PRIMARY KEY,
  secret_hash   TEXT NOT NULL,              -- 匿名認証用
  display_name  TEXT NOT NULL,
  island_id     TEXT NOT NULL,
  last_pos_x    REAL NOT NULL,
  last_pos_y    REAL NOT NULL,
  created_at    INTEGER NOT NULL,
  last_seen_at  INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS pet (
  id            INTEGER PRIMARY KEY,        -- EntityId
  player_id     TEXT NOT NULL UNIQUE REFERENCES player(id),
  species       TEXT NOT NULL,              -- mofi/mizune/hakka/momona/hoshira
  name          TEXT NOT NULL,
  persona_json  TEXT NOT NULL,              -- 性格タグ・口ぐせ・好物・苦手
  traits_json   TEXT NOT NULL,
  affection     REAL NOT NULL DEFAULT 30,
  summary       TEXT NOT NULL DEFAULT '',   -- 長期記憶の圧縮テキスト
  created_at    INTEGER NOT NULL
);

-- 島の出来事（誰の記憶にもなりうる客観ログ）
CREATE TABLE IF NOT EXISTS island_event (
  id            INTEGER PRIMARY KEY AUTOINCREMENT,
  island_id     TEXT NOT NULL,
  tick          INTEGER NOT NULL,
  island_day    INTEGER NOT NULL,
  kind          TEXT NOT NULL,              -- born/died/quarrel/befriend/harvest/build/weather...
  actor_id      INTEGER,
  target_id     INTEGER,
  pos_x         REAL, pos_y REAL,
  text          TEXT NOT NULL,              -- 「ミズネがハッカとケンカした」
  importance    INTEGER NOT NULL            -- 1..10
);
CREATE INDEX IF NOT EXISTS idx_event_day ON island_event(island_id, island_day);

-- ペットのエピソード記憶（主観。island_eventから複写 or 会話から生成）
CREATE TABLE IF NOT EXISTS pet_memory (
  id            INTEGER PRIMARY KEY AUTOINCREMENT,
  pet_id        INTEGER NOT NULL REFERENCES pet(id),
  tick          INTEGER NOT NULL,
  island_day    INTEGER NOT NULL,
  kind          TEXT NOT NULL,              -- talk/observe/gossip/diary
  text          TEXT NOT NULL,
  keywords      TEXT NOT NULL,              -- 空白区切り。検索用
  importance    INTEGER NOT NULL,           -- 1..10
  last_access_tick INTEGER NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_mem_pet ON pet_memory(pet_id, tick DESC);

-- 会話ログ（プレイヤー↔ペット、ペット↔ペット）
CREATE TABLE IF NOT EXISTS chat_log (
  id            INTEGER PRIMARY KEY AUTOINCREMENT,
  island_id     TEXT NOT NULL,
  tick          INTEGER NOT NULL,
  speaker_kind  TEXT NOT NULL,              -- player/pet/critter
  speaker_id    TEXT NOT NULL,
  listener_id   TEXT,
  text          TEXT NOT NULL
);

-- 個体間の好感度（動物含む。上限を超えたら弱い関係から削除）
CREATE TABLE IF NOT EXISTS relation (
  a_id          INTEGER NOT NULL,
  b_id          INTEGER NOT NULL,
  score         REAL NOT NULL,              -- -100..100
  met_count     INTEGER NOT NULL DEFAULT 0,
  PRIMARY KEY (a_id, b_id)
);

-- LLM使用量（コスト監視とレート制限の永続化）
CREATE TABLE IF NOT EXISTS llm_usage (
  id            INTEGER PRIMARY KEY AUTOINCREMENT,
  ts            INTEGER NOT NULL,
  player_id     TEXT,
  purpose       TEXT NOT NULL,              -- dialogue/decide/diary/gossip
  prompt_tokens INTEGER NOT NULL,
  completion_tokens INTEGER NOT NULL,
  latency_ms    INTEGER NOT NULL,
  ok            INTEGER NOT NULL
);
```

## 4. 永続化タイミング

| データ | 書き込み契機 |
|---|---|
| `island`（tick/日/季節/天気） | 30秒ごと＋停止時 |
| `island_snapshot` | 30秒ごと（トランザクション1回）＋停止時 |
| `player` の位置・last_seen | 切断時＋30秒ごと |
| `pet`（懐き度・summary） | 変化時（デバウンス5秒） |
| `island_event` / `pet_memory` / `chat_log` | 発生時に即書き込み |
| `llm_usage` | 各LLM呼び出し完了時 |

## 5. 容量見積り

- `island_snapshot`: 動物120体 × 約400B ＋ tiles_decay 16KB ≒ **70KB**（1行を上書き）
- `island_event`: 1島日あたり 200件程度 × 120B ≒ 24KB/日 → **9MB/年（実時間15日）**
- `pet_memory`: 1ペット1島日あたり 30件 → 16プレイヤーで 60KB/日

→ 1ファイル数十MBで収まる。`island_event` は90島日より古い `importance <= 3` を日次で削除する。
